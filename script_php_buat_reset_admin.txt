<?php
require_once __DIR__ . '/config/database.php';

$database = new Database();
$db = $database->getConnection();

$username = 'admin';
$password = 'admin123';
$hashedPassword = password_hash($password, PASSWORD_DEFAULT);
$nama = 'Administrator';

// Update jika sudah ada, atau insert baru
$stmt = $db->prepare("SELECT id FROM users WHERE username = :username");
$stmt->execute([':username' => $username]);

if ($stmt->fetch()) {
    $update = $db->prepare("UPDATE users SET password = :password, nama = :nama WHERE username = :username");
    $update->execute([
        ':password' => $hashedPassword,
        ':nama'     => $nama,
        ':username' => $username
    ]);
    echo "<h2>✅ Password admin BERHASIL direset menjadi: <u>admin123</u></h2>";
} else {
    $insert = $db->prepare("INSERT INTO users (username, password, nama) VALUES (:username, :password, :nama)");
    $insert->execute([
        ':username' => $username,
        ':password' => $hashedPassword,
        ':nama'     => $nama
    ]);
    echo "<h2>✅ Akun admin BERHASIL dibuat dengan password: <u>admin123</u></h2>";
}

echo '<p><a href="login.php">Klik di sini untuk langsung Login</a></p>';
